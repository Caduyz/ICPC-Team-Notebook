\subsection{Contagem de Inversões}

Um par $(i,j)$ é uma inversão se:
\[
i < j \quad\text{e}\quad a_i > a_j
\]

\textbf{Complexidade:} $O(N\log N)$.

\begin{lstlisting}
long long count_inversions (vector<int>& a) {
    // Compressao de Coordenadas
    vector<int> vals = a;
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());
    SegTree st(vals.size()); // Segtree para soma e update = 1
    long long inv = 0;

    for (int x : a) {
        int rank = lower_bound(vals.begin(), vals.end(), x) - vals.begin();
        inv += st.query(rank + 1, (int)vals.size() - 1); 
        st.update(rank, 1);
    }
    return inv;
}
\end{lstlisting}
