\section{Grafos}



\subsection{MST --- Kruskal}

\textbf{Complexidade:} $O(E\log E)$.

\begin{lstlisting}
using Edge = tuple<long long, int, int>;
long long kruskal (int n, vector<Edge> edges) {
    priority_queue<Edge, vector<Edge>, greater<Edge>> pq;
    for (const auto& [w, u, v] : edges) pq.push({w, u, v});
    DSU dsu(n);
    long long totW = 0;
    int used = 0;
    while (!pq.empty()) {
        auto [w, u, v] = pq.top(); pq.pop();
        if (!dsu.unite(u, v)) continue;
        totW += w;
        used++;
        if (used == n - 1) break;
    }
    if (used != n - 1)
        return -1;
    return totW;
}
\end{lstlisting}


\subsection{MST --- Prim}

Adequado para grafos densos representados por matriz de adjacência.

\textbf{Complexidade:} $O(V^2)$.

\begin{lstlisting}
long long prim (const vector<vector<long long>>& graph) {
    int n = graph.size();
    vector<long long> minEdge(n, INF);
    vector<bool> used(n, false);
    long long totW = 0;
    minEdge[0] = 0;

    for (int i = 0; i < n; i++) {
        int curr = -1;
        for (int j = 0; j < n; j++)
            if (!used[j] && (curr == -1 || minEdge[j] < minEdge[curr])) curr = j;
        if (curr == -1 || minEdge[curr] == INF) return -1;
        used[curr] = true;
        totW += minEdge[curr];
        for (int k = 0; k < n; k++)
            if (!used[k] && graph[curr][k] < minEdge[k]) minEdge[k] = graph[curr][k];
    }
    return totW;
}
\end{lstlisting}



\subsection{Bellman-Ford}

Encontra menores caminhos a partir de uma origem em grafos com pesos negativos.

Também detecta ciclos negativos alcançáveis a partir da origem.

\textbf{Complexidade:} $O(VE)$.

\begin{lstlisting}
using Edge = tuple<long long, int, int>;
pair<vector<long long>, bool> bellmanFord (int n, const vector<Edge>& edges, int origin) {
    vector<long long> dist(n, INF);
    dist[origin] = 0;
    for (int i = 0; i < n - 1; i++) {
        bool changed = false;
        for (auto [u, v, w] : edges) {
            if (dist[u] == INF) continue;
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                changed = true;
            }
        }
        if (!changed) break;
    }
    for (const Edge& edge : edges) {
        if (dist[u] == INF) continue;
        if (dist[u] + w < dist[v]) return {dist, true};
    }
    return {dist, false};
}
\end{lstlisting}


\subsection{Floyd-Warshall}

Encontra os menores caminhos entre \textbf{todos os pares}
de vértices.

Também funciona com arestas negativas, desde que não existam
ciclos negativos.

\textbf{Complexidade:} $O(V^3)$.

\textbf{Memória:} $O(V^2)$.

\begin{lstlisting}
void floydWarshall (vector<vector<long long>>& dist) {
    int n = dist.size();
    for (int k = 0; k < n; k++) {
        for (int u = 0; u < n; u++) {
            if (dist[u][k] == INF) continue;
            for (int v = 0; v < n; v++) {
                if (dist[k][v] == INF) continue;
                dist[u][v] = min(dist[u][v], dist[u][k] + dist[k][v]);
            }
        }
    }
}
\end{lstlisting}

Após o algoritmo, se:
\[
dist[v][v] < 0
\]

para algum vértice $v$, existe um ciclo negativo.
