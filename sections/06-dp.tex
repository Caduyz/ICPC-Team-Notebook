\section{Programação Dinâmica}



\subsection{LIS --- Longest Increasing Subsequence}

\textbf{Complexidade:} $O(N\log N)$.

\begin{lstlisting}
int LIS (const vector<int>& values) {
    vector<int> tails;
    for (int value : values) {
        auto position = lower_bound(tails.begin(), tails.end(), value);
        if (position == tails.end()) tails.push_back(value);
        else *position = value;
    }
    return tails.size();
}
\end{lstlisting}

\texttt{tails[i]} representa o menor valor possível para o
último elemento de uma LIS de tamanho $i+1$.

Para subsequência \textbf{não decrescente}, use \texttt{upper\_bound}.


\subsection{LCS --- Longest Common Subsequence}

\textbf{Complexidade:} $O(NM)$.

\begin{lstlisting}
int LCS (const string& first, const string& second) {
    int fLen = first.size();
    int sLen = second.size();
    vector<vector<int>> dp(fLen + 1, vector<int>(sLen + 1, 0));
    for (int i = 1; i <= fLen; i++) {
        for (int j = 1; j <= sLen; j++) {
            if (first[i - 1] == second[j - 1]) dp[i][j] = dp[i - 1][j - 1] + 1;
            else dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
        }
    }
    return dp[fLen][sLen];
}
\end{lstlisting}

Transição:
\[
dp[i][j] =
\begin{cases}
dp[i-1][j-1] + 1,
& a_i=b_j \\[4pt]
\max(dp[i-1][j],dp[i][j-1]),
& a_i\neq b_j
\end{cases}
\]


\subsection{Maximum Sum 2D}

\textbf{Complexidade:} $O(N^2M)$.

Para uma matriz $N\times M$:

\begin{lstlisting}
long long maximumSum2D (const vector<vector<long long>>& matrix) {
    int n = matrix.size();
    int m = matrix[0].size();
    long long ans = LLONG_MIN;

    for (int i = 0; i < n; i++) {
        vector<long long> columnSum(m, 0);
        for (int j = i; j < n; j++) {
            for (int column = 0; column < m; column++)
                columnSum[column] += matrix[j][column];
            long long currentSum = 0;
            for (long long value : columnSum) {
                currentSum = max(value, currentSum + value);
                ans = max(ans, currentSum);
            }
        }
    }
    return ans;
}
\end{lstlisting}
