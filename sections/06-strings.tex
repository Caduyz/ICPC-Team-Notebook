\section{Strings}

\subsection{KMP}

\textbf{Prefix Function:} $O(m)$.

\textbf{Busca:} $O(n)$.

\begin{lstlisting}
vector<int> prefixFunction (const string& pattern) {
    int pLen = pattern.size();
    vector<int> pref(pLen);
    for (int i = 1; i < pLen; i++) {
        int j = pref[i - 1];
        while (j > 0 && pattern[i] != pattern[j]) 
            j = pref[j - 1];
        if (pattern[i] == pattern[j]) j++;
        pref[i] = j;
    }
    return pref;
}
int kmp (const string& text, const string& pattern) {
    vector<int> pref = prefFunction(pattern);
    int pLen = pattern.size();
    for (int i = 0, j = 0; i < text.size(); i++) {
        while (j > 0 && text[i] != pattern[j])
            j = pref[j - 1];
        if (text[i] == pattern[j]) j++;
        if (j == pLen) return i - pLen + 1;
    }
    return -1;
}
\end{lstlisting}


\subsection{Z-Function}

\textbf{Complexidade:} $O(n)$.

\begin{lstlisting}
vector<int> zFunction (const string& text) {
    int n = text.size();
    vector<int> z(n);
    int l = 0, r = 0;
    for (int i = 1; i < n; i++) {
        if (i <= r) z[i] = min(r - i + 1, z[i - l]);
        while (i + z[i] < n && text[z[i]] == text[i + z[i]]) z[i]++;
        if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;
    }
    return z;
}
\end{lstlisting}


\subsection{Polynomial Rolling Hash}

Utiliza double hashing para evitar colisões.

\textbf{Complexidade:} $O(n)$ para Pré-processamento.

\begin{lstlisting}
struct DoubleHash {
    ll h1, h2;
    bool operator==(const DoubleHash& other) const {
        return h1 == other.h1 && h2 == other.h2;
    }
    bool operator<(const DoubleHash& other) const {
        if (h1 != other.h1) return h1 < other.h1;
        return h2 < other.h2;
    }
};
struct RollingHash {
    ll MOD1 = 1e9 + 7, MOD2 = 1e9 + 9;
    ll P1 = 31, P2 = 37;
    vector<ll> h1, h2, p1, p2;
    
    RollingHash(const string& s) {
        int n = s.size();
        h1.assign(n + 1, 0); h2.assign(n + 1, 0);
        p1.assign(n + 1, 1); p2.assign(n + 1, 1);
        for (int i = 0; i < n; i++) {
            p1[i+1] = (p1[i] * P1) % MOD1;
            p2[i+1] = (p2[i] * P2) % MOD2;
            h1[i+1] = (h1[i] * P1 + (s[i] - 'a' + 1)) % MOD1;
            h2[i+1] = (h2[i] * P2 + (s[i] - 'a' + 1)) % MOD2;
        }
    }
    DoubleHash get(int l, int r) { // 0-indexed, inclusive
        ll hash1 = (h1[r+1] - h1[l] * p1[r - l + 1]) % MOD1;
        if (hash1 < 0) hash1 += MOD1;
        ll hash2 = (h2[r+1] - h2[l] * p2[r - l + 1]) % MOD2;
        if (hash2 < 0) hash2 += MOD2;
        return {hash1, hash2};
    }
};
\end{lstlisting}


\subsection{Manacher's Algorithm}

Encontra o maior palíndromo centrado em cada posição (ímpar e par) de uma string.

\textbf{Complexidade:} $O(n)$.

\begin{lstlisting}
vector<vector<int>> manacher(string s) {
    int n = s.size();
    vector<vector<int>> d(2, vector<int>(n, 0));
    for (int z = 0; z < 2; z++) {
        for (int i = 0, l = 0, r = -1; i < n; i++) {
            int k = (i > r) ? z : min(d[z][l + r - i + z], r - i + 1);
            while (0 <= i - k - z && i + k < n && s[i - k - z] == s[i + k]) k++;
            d[z][i] = k--;
            if (i + k > r) {
                l = i - k - z;
                r = i + k;
            }
        }
    }
    return d;
}
\end{lstlisting}


\subsection{Aho-Corasick Automaton}

\textbf{Complexidade:} $O(n + m + \text{matches})$.

\begin{lstlisting}
const int MAXC = 26;
struct AhoCorasick {
    struct Node {
        int next[MAXC];
        int fail = 0;
        int elink = 0;
        vector<int> pids;
        Node() { memset(next, -1, sizeof(next)); }
    };
 
    vector<Node> trie;
    AhoCorasick() {
        trie.emplace_back(); 
    }
    void insert(const string& s, int id) {
        int u = 0;
        for (char c : s) {
            int idx = c - 'a';
            if (trie[u].next[idx] == -1) {
                trie[u].next[idx] = trie.size();
                trie.emplace_back();
            }
            u = trie[u].next[idx];
        }
        trie[u].pids.pb(id);
    }
    void build() {
        queue<int> q;
        for (int c = 0; c < MAXC; ++c) {
            if (trie[0].next[c] != -1) q.push(trie[0].next[c]);
            else trie[0].next[c] = 0;
        }
        while (sz(q)) {
            int u = q.front(); q.pop();
            for (int c = 0; c < MAXC; ++c) {
                int v = trie[u].next[c];
                if (v != -1) {
                    trie[v].fail = trie[trie[u].fail].next[c];
                    int f = trie[v].fail;
                    if (!trie[f].pids.empty()) trie[v].elink = f;
                    else trie[v].elink = trie[f].elink;
                    q.push(v);
                } 
                else trie[u].next[c] = trie[trie[u].fail].next[c];
            }
        }
    }
    void search(const string& s, vector<vector<int>>& occur) {
        int n = sz(s);
        int u = 0;
        for (int i = 0; i < n; i++) {
            int idx = s[i] - 'a';
            u = trie[u].next[idx];
            int curr = u;
            while (curr > 0) {
                for (int id : trie[curr].pids) occur[id].pb(i);
                curr = trie[curr].elink;
            }
        }
    }
};
\end{lstlisting}


\subsection{Suffix Automaton (SAM)}

\textbf{Complexidade:} $O(n)$.

\begin{lstlisting}
struct State {
    int len, link;
    int next[26]; // padrao para letras de 'a' a 'z', reajuste se necessario
};

const int MAXLEN = 1e5 + 5; // ajuste conforme o tamanho maximo esperado
State st[MAXLEN * 2];
int sz, last;

void sam_init() {
    st[0].len = 0;
    st[0].link = -1;
    memset(st[0].next, -1, sizeof(st[0].next));
    sz = 1;
    last = 0;
}
void sam_extend(char c) {
    int id = c - 'a'; // intervalo [0, 25], reajuste se necessario
    int cur = sz++;
    st[cur].len = st[last].len + 1;
    memset(st[cur].next, -1, sizeof(st[cur].next));
    
    int p = last;
    while (p != -1 && st[p].next[id] == -1) {
        st[p].next[id] = cur;
        p = st[p].link;
    }
    
    if (p == -1)
        st[cur].link = 0;
    else {
        int q = st[p].next[id];
        if (st[p].len + 1 == st[q].len) {
            st[cur].link = q;
        } else {
            int clone = sz++;
            st[clone].len = st[p].len + 1;
            memcpy(st[clone].next, st[q].next, sizeof(st[q].next)); 
            st[clone].link = st[q].link;
            while (p != -1 && st[p].next[id] == q) {
                st[p].next[id] = clone;
                p = st[p].link;
            }
            st[q].link = st[cur].link = clone;
        }
    }
    last = cur;
}
\end{lstlisting}
