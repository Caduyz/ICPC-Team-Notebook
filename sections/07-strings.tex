\section{Strings}

\subsection{KMP}

\textbf{Prefix Function:} $O(M)$.

\textbf{Busca:} $O(N)$.

\begin{lstlisting}
vector<int> prefixFunction (const string& pattern) {
    int pLen = pattern.size();
    vector<int> pref(pLen);
    for (int i = 1; i < pLen; i++) {
        int matched = pref[i - 1];
        while (matched > 0 && pattern[i] != pattern[matched]) 
            matched = pref[matched - 1];
        if (pattern[i] == pattern[matched]) matched++;
        pref[i] = matched;
    }
    return pref;
}
int kmp (const string& text, const string& pattern) {
    vector<int> pref = prefFunction(pattern);
    int pLen = pattern.size();
    for (int txtIdx = 0, matched = 0; txtIdx < text.size(); txtIdx++) {
        while (matched > 0 && text[txtIdx] != pattern[matched])
            matched = pref[matched - 1];
        if (text[txtIdx] == pattern[matched]) matched++;
        if (matched == pLen) return txtIdx - pLen + 1;
    }
    return -1;
}
\end{lstlisting}



\subsection{Z-Function}

\textbf{Complexidade:} $O(N)$.

\begin{lstlisting}
vector<int> zFunction (const string& text) {
    int n = text.size();
    vector<int> z(n);
    int l = 0, r = 0;
    for (int i = 1; i < n; i++) {
        if (i <= r) z[i] = min(r - i + 1, z[i - l]);
        while (i + z[i] < n && text[z[i]] == text[i + z[i]]) z[i]++;
        if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;
    }
    return z;
}
\end{lstlisting}
